## Executive Summary

ClinSight is a **technical demonstration**, not a clinical deployment. All data used in the hackathon MVP falls into three categories:

1. **Public de-identified research datasets** (NIH Chest X-ray14, RSNA) — images
2. **Hand-crafted synthetic data** (labs, triage notes, demographics) — clinically reviewed
3. **Synthetic patient generators** (Synthea) — for supplementary test cases

**No real patient data is used. No protected health information (PHI) is processed. No HIPAA-covered entities are involved.**

**Preprocessing Script:**
```python
# scripts/preprocess_nih.py

import os
import numpy as np
from PIL import Image
from pathlib import Path

def preprocess_nih_image(image_path: str, output_dir: str = "data/processed"):
    """
    NIH preprocessing pipeline:
    1. Load grayscale PNG
    2. Convert to RGB (required for VLM)
    3. Resize to 512×512 (model input size)
    4. Normalize to [0, 1]
    5. Save as PNG
    """
    # Load
    img = Image.open(image_path).convert('RGB')
    
    # Resize (LANCZOS for quality)
    img_resized = img.resize((512, 512), Image.LANCZOS)
    
    # Normalize (save as float PNG or keep uint8)
    img_array = np.array(img_resized)
    
    # Save processed
    output_path = Path(output_dir) / Path(image_path).name
    Image.fromarray(img_array).save(output_path, 'PNG')
    
    return output_path

def batch_preprocess_nih(input_dir: str = "data/nih/images", 
                          output_dir: str = "data/nih/processed"):
    os.makedirs(output_dir, exist_ok=True)
    
    image_files = sorted(Path(input_dir).glob("*.png"))
    print(f"Processing {len(image_files)} images...")
    
    for img_path in image_files:
        preprocess_nih_image(str(img_path), output_dir)
    
    print(f"Done. Processed images saved to {output_dir}")

if __name__ == "__main__":
    batch_preprocess_nih()
```

```python
# scripts/extract_synthea_labs.py

import pandas as pd
from pathlib import Path

def extract_labs_from_synthea(output_dir: str = "./output"):
    """
    Synthea generates observations.csv with all lab values.
    We filter for relevant ED labs only.
    """
    obs_path = Path(output_dir) / "observations.csv"
    obs = pd.read_csv(obs_path)
    
    # Filter for emergency-relevant labs
    relevant_codes = {
        "2339-0": "glucose",
        "718-7": "wbc",
        "33717-0": "platelets", 
        "20570-8": "hemoglobin",
        "38483-4": "creatinine",
        "2947-0": "sodium",
        "2823-3": "potassium",
        "1963-8": "bicarbonate",
        "2703-7": "pO2",
        "2019-8": "pCO2",
        "2744-1": "pH",
        "2532-0": "lactate",
        "10839-9": "troponin",
        "33762-6": "NT_pro_BNP",
        "1751-7": "albumin"
    }
    
    filtered = obs[obs["CODE"].isin(relevant_codes.keys())]
    filtered["LAB_NAME"] = filtered["CODE"].map(relevant_codes)
    
    return filtered[["PATIENT", "LAB_NAME", "VALUE", "UNITS", "DATE"]]
```

### 5.2 Case Packet Merger Script

```python
# scripts/build_case_packet.py

import json
from pathlib import Path
from typing import Dict, Any

def build_case_packet(
    case_id: str,
    image_path: str,
    labs: Dict[str, Any],
    triage_note: Dict[str, Any],
    demographics: Dict[str, Any]
) -> Dict[str, Any]:
    """
    Merges all data sources into a single case packet.
    This is the input to the ClinSight agent system.
    """
    
    packet = {
        "case_id": case_id,
        "timestamp": "2026-05-10T14:30:00Z",
        "patient": demographics,
        "input": {
            "image": {
                "path": image_path,
                "modality": "CXR",
                "view": "PA",
                "source": "NIH-ChestXray14"
            },
            "labs": {
                "values": labs,
                "units": {
                    "wbc": "cells/uL",
                    "pO2": "mmHg",
                    "pCO2": "mmHg",
                    "pH": "unitless",
                    "lactate": "mmol/L",
                    "troponin": "ng/mL",
                    "hemoglobin": "g/dL",
                    "platelets": "cells/uL",
                    "creatinine": "mg/dL",
                    "glucose": "mg/dL",
                    "sodium": "mEq/L",
                    "potassium": "mEq/L",
                    "bicarbonate": "mEq/L",
                    "albumin": "g/dL",
                    "NT_pro_BNP": "pg/mL"
                }
            },
            "clinical_note": triage_note
        },
        "metadata": {
            "data_version": "1.0",
            "synthetic": True,
            "phi_free": True,
            "reviewed_by": "Clinical Advisor (EM Resident)"
        }
    }
    
    return packet

def build_all_packets(output_dir: str = "data/case_packets"):
    """Build all 6 demo case packets."""
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    
    # Load sources
    with open("data/labs/6_case_labs.json") as f:
        all_labs = json.load(f)
    
    with open("data/notes/6_case_notes.json") as f:
```

### 5.3 Preprocessing Validation Rules

```python
# scripts/validate_packet.py

import json
from pathlib import Path

def validate_case_packet(packet_path: str) -> list:
    """Validate a case packet before it enters the system."""
    errors = []
    
    with open(packet_path) as f:
        packet = json.load(f)
    
    # Required fields
    required = ["case_id", "patient", "input", "metadata"]
    for field in required:
        if field not in packet:
            errors.append(f"Missing required field: {field}")
    
    # Image exists
    img_path = Path(packet["input"]["image"]["path"])
    if not img_path.exists():
        errors.append(f"Image not found: {img_path}")
    
    # Labs have expected ranges (basic sanity)
    labs = packet["input"]["labs"]["values"]
    if labs.get("pH", 7.4) < 6.8 or labs.get("pH", 7.4) > 7.8:
        errors.append("pH out of survivable range")
    
    if labs.get("sodium", 140) < 100 or labs.get("sodium", 140) > 180:
        errors.append("Sodium out of survivable range")
    
    # Metadata flags
    if not packet["metadata"].get("synthetic", False):
        errors.append("WARNING: Packet not marked as synthetic")
    
    if not packet["metadata"].get("phi_free", False):
        errors.append("WARNING: Packet not marked as PHI-free")
    
    return errors
```
